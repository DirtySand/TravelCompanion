cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
include(pico_extras_import.cmake)
project(travel-companion C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

add_subdirectory(drivers/bme680)
add_subdirectory(drivers/SSD1351U3)
add_subdirectory(drivers/ds3231)
add_library(timezone STATIC timezone.c)
add_library(datetime_utils STATIC datetime_utils.c)
target_link_libraries(datetime_utils PUBLIC ds3231)
add_library(power STATIC power.c)
target_link_libraries(power PUBLIC hardware_sleep hardware_gpio pico_stdlib hardware_clocks)
add_executable(travel-companion main.c)
target_link_libraries(travel-companion pico_stdlib hardware_i2c bme680_port ssd1351_port ds3231_port timezone datetime_utils power hardware_clocks)
add_library(board_config INTERFACE)
target_include_directories(board_config INTERFACE ${CMAKE_CURRENT_LIST_DIR})

pico_enable_stdio_usb(travel-companion 1)
pico_enable_stdio_uart(travel-companion 0)
pico_add_extra_outputs(travel-companion)
